  (function(){
    var audio = document.getElementById('radioAudio');
    var play = document.getElementById('playButton');
    var topPlay = document.getElementById('topPlay');
    var status = document.getElementById('status');
    var volume = document.getElementById('volume');
    var mute = document.getElementById('muteButton');

    function playing(){
      play.textContent = '❚❚';
      topPlay.textContent = '❚❚';
      status.textContent = '● Reproduciendo VK Radio en vivo';
    }
    function paused(){
      play.textContent = '▶';
      topPlay.textContent = '▶';
      status.textContent = '● Pausado';
    }
    function toggle(){
      if(audio.paused){
        var promise = audio.play();
        if(promise && promise.then){
          promise.then(playing).catch(function(){
            status.textContent = 'No se pudo conectar. Comprueba la señal de Zeno e inténtalo otra vez.';
          });
        }
      }else{
        audio.pause();
        paused();
      }
    }
    play.addEventListener('click',toggle);
    topPlay.addEventListener('click',toggle);
    volume.addEventListener('input',function(){
      audio.volume = Number(volume.value);
      audio.muted = audio.volume === 0;
      mute.textContent = audio.muted ? '🔇' : '🔊';
    });
    mute.addEventListener('click',function(){
      audio.muted = !audio.muted;
      mute.textContent = audio.muted ? '🔇' : '🔊';
    });
    document.getElementById('restartButton').addEventListener('click',function(){
      audio.pause();
      paused();
    });
    document.getElementById('liveButton').addEventListener('click',function(){
      if(audio.paused){toggle();}
      else{status.textContent='● Estás escuchando la señal en vivo';}
    });
    audio.addEventListener('playing',playing);
    audio.addEventListener('pause',function(){if(audio.paused){paused();}});
    audio.volume = 0.8;

    function makeBars(id,count){
      var parent = document.getElementById(id);
      for(var i=0;i<count;i++){
        var bar = document.createElement('i');
        bar.style.height = (10 + Math.random()*85) + '%';
        bar.style.animationDuration = (0.3 + Math.random()*0.7) + 's';
        parent.appendChild(bar);
      }
    }
    makeBars('topEq',55);
    makeBars('miniEq',32);
  })();
